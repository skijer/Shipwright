"""Regenerates catalog.h from ModLoader64's lists: https://modloader64.com/assets/data/models.json and audio.json.

Usage: python gen_catalog.py catalog.h models.json [audio.json ...]
"""
import json
import sys
from pathlib import PurePosixPath


def c_string(text):
    # Octal escapes keep UTF-8 intact without MSVC/Clang source-charset flags.
    out = []
    for byte in text.encode("utf-8"):
        char = chr(byte)
        if byte < 0x20 or byte >= 0x7F or char in '"\\?':
            out.append("\\%03o" % byte)
        else:
            out.append(char)
    return '"' + "".join(out) + '"'


def author_names(model):
    names = []
    for author in model.get("authors", []):
        names.append(author["name"] if isinstance(author, dict) else author)
    return ", ".join(name for name in names if name)


def main(target, sources):
    rows = {}
    for source in sources:
        with open(source, encoding="utf-8") as handle:
            data = json.load(handle)
        for model in data.get("models", []) + data.get("mods", []):
            file = PurePosixPath(model["download"]).name.lower()
            rows.setdefault(file, (model["name"].strip(), author_names(model)))
    lines = [
        "#pragma once",
        "// Generated by gen_catalog.py from modloader64.com/assets/data/models.json and audio.json.",
        "namespace pak {",
        "struct CatalogEntry { const char* file; const char* name; const char* author; };",
        "static constexpr CatalogEntry kCatalog[] = {",
    ]
    for file, (name, author) in sorted(rows.items()):
        lines.append("    {%s, %s, %s}," % (c_string(file), c_string(name), c_string(author)))
    lines += ["};", "}", ""]
    with open(target, "w", encoding="ascii", newline="\n") as handle:
        handle.write("\n".join(lines))


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2:])
