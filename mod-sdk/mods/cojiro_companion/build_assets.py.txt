"""Clone cucco materials locally; tint only the three white feather textures."""
from pathlib import Path
import struct
import zipfile
import json
from collections import deque

HERE=Path(__file__).resolve().parent
ROOT=HERE.parents[3]
ARCHIVE=ROOT/'Shipwright-unbound/x64/Release/oot.o2r'
OUT=HERE/'assets/objects/object_nei_cojiro';OUT.mkdir(parents=True,exist_ok=True)
def crc(path):
    value=(1<<64)-1
    for byte in path.encode():
        value^=byte<<56
        for _ in range(8):value=((value<<1)^ (0x42F0E1EBA9EA3693 if value>>63 else 0))&((1<<64)-1)
    return value
def text(data,pos):
    n,=struct.unpack_from('<I',data,pos);pos+=4
    return data[pos:pos+n].decode().removeprefix('__OTR__'),pos+n

with zipfile.ZipFile(ARCHIVE) as z:
    remap={};report={'textures':{},'display_lists':{}}
    for name in ('gCuccoBaseFeathersTex','gCuccoHeadFeathersTex','gCuccoTailFeathersTex','gCuccoEyeTex'):
        path='objects/object_niw/'+name;raw=bytearray(z.read(path));kind,w,h,n=struct.unpack_from('<IIII',raw,64)
        assert kind==2 and n==w*h*2
        mask=None
        if name=='gCuccoEyeTex':
            # The original white feather background forms a visible square on a blue bird.
            # Flood only neutral, light border-connected pixels. Keep the iris, pupil and highlights intact.
            pixels=[]
            for pos in range(80,len(raw),2):
                pixel,=struct.unpack_from('>H',raw,pos)
                pixels.append(((pixel>>11)&31,(pixel>>6)&31,(pixel>>1)&31))
            mask=set();queue=deque((x,y) for y in range(h) for x in range(w) if x in (0,w-1) or y in (0,h-1))
            while queue:
                x,y=queue.popleft()
                if not (0<=x<w and 0<=y<h) or (x,y) in mask:continue
                rgb=pixels[y*w+x]
                if min(rgb)<21 or max(rgb)-min(rgb)>2:continue
                mask.add((x,y));queue.extend(((x-1,y),(x+1,y),(x,y-1),(x,y+1)))
            report['eye_background_pixels']=len(mask)
        for pos in range(80,len(raw),2):
            index=(pos-80)//2
            if mask is not None and (index%w,index//w) not in mask:continue
            pixel,=struct.unpack_from('>H',raw,pos)
            r,g,b=(pixel>>11)&31,(pixel>>6)&31,(pixel>>1)&31
            light=max(r,g,b)
            rgb=[round(light*component/255) for component in (55,110,245)]
            struct.pack_into('>H',raw,pos,(rgb[0]<<11)|(rgb[1]<<6)|(rgb[2]<<1)|(pixel&1))
        (OUT/name).write_bytes(raw);remap[crc(path)]=crc('objects/object_nei_cojiro/'+name)
        report['textures'][name]=[w,h]
    names=('gCuccoTailAndUndersideDL','gCuccoLeftFootDL','gCuccoLeftWingDL','gCuccoRightFootDL',
           'gCuccoRightWingDL','gCuccoNeckDL','gCuccoHeadDL')
    for name in names:
        path='objects/object_niw/'+name;raw=bytearray(z.read(path));pos=72;count=0
        while pos<len(raw):
            w0,w1=struct.unpack_from('<II',raw,pos);op=w0>>24
            if op in (0x20,0x31,0x32,0x04,0x33,0x36,0x37):
                a,b=struct.unpack_from('<II',raw,pos+8);hash=(a<<32)|b
                if op==0x20 and hash in remap:
                    target=remap[hash];struct.pack_into('<II',raw,pos+8,target>>32,target&0xffffffff);count+=1
                pos+=16
            else:pos+=8
            if op==0xdf:break
        assert pos==len(raw),(name,pos,len(raw))
        (OUT/name).write_bytes(raw);report['display_lists'][name]=count
    assert sum(report['display_lists'].values())>=3,report
    # Native skeleton limb indices select the local copies without touching shared resources.
    raw=z.read('objects/object_niw/gCuccoSkel');pos=64+10+1
    count,=struct.unpack_from('<I',raw,pos);pos+=4
    limbs=[]
    for _ in range(count):name,pos=text(raw,pos);limbs.append(name)
    mapping=[]
    for index,name in enumerate(limbs):
        raw=z.read(name)
        for dl in names:
            if dl.encode() in raw:mapping.append((index+1,dl));break
    assert len(mapping)==7,mapping
    (HERE/'limb_models.inc.c').write_text('/* Native cucco limb indices; generated by build_assets.py. */\n'+
        ''.join(f'static const ALIGN_ASSET(2) char sLimb{index}[]="__OTR__objects/object_nei_cojiro/{dl}";\n' for index,dl in mapping)+
        'static const struct { s32 limb; const char* path; } sModels[]={\n'+
        ''.join(f'    {{{index},sLimb{index}}},\n' for index,dl in mapping)+'};\n')
    report['limbs']=mapping
(HERE/'asset-report.json.txt').write_text(json.dumps(report,indent=2)+'\n')
print('Cojiro: local native geometry; only white feather material references tinted blue:',mapping)
