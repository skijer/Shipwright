"""Bake Romani's CC0 companion clips onto both native OoT Malon skeletons.

Uses the first idle pose as a reference in each rig. Animation deltas are
transferred in model space, then resolved into the target hierarchy. Assets are
ordinary OTR Animation resources, so no retarget work runs in the game.
"""
from pathlib import Path
import argparse
import json
import re
import struct
import zipfile
import math
import numpy as np

HERE = Path(__file__).resolve().parent
SOURCE_COMMIT = 'cced2388ea8c8a9082346ce70e227d805162a8ee'

class Reader:
    def __init__(self, data): self.data, self.pos = data, 64
    def read(self, fmt):
        result = struct.unpack_from('<' + fmt, self.data, self.pos)
        self.pos += struct.calcsize('<' + fmt)
        return result
    def string(self):
        n, = self.read('I')
        value = self.data[self.pos:self.pos+n].decode()
        self.pos += n
        return value

def load_skeleton(archive, path):
    reader = Reader(archive.read(path))
    reader.read('BBII'); reader.read('B')
    count, = reader.read('I')
    names = [reader.string() for _ in range(count)]
    limbs = []
    for name in names:
        reader = Reader(archive.read(name))
        reader.read('BB'); reader.string(); reader.read('HI'); reader.string()
        reader.read('fffHHH')
        for _ in range(4): reader.string()
        x, y, z, child, sibling = reader.read('hhhBB')
        limbs.append(dict(name=name, pos=[x, y, z], child=child, sibling=sibling))
    return limbs

def load_animation(data):
    reader = Reader(data)
    kind, frames, n = reader.read('IhI')
    assert kind == 0
    values = reader.read('h' * n)
    count, = reader.read('I')
    indices = [reader.read('HHH') for _ in range(count)]
    static, = reader.read('H')
    return np.array([[[values[i if i < static else i+f] for i in joint]
                      for joint in indices] for f in range(frames)], dtype=float)

def source_data():
    text = (HERE/'reference/gRomaniSkel.c.txt').read_text()
    arrays = {}
    for name, body in re.findall(r'(?:s16|JointIndex)\s+(\w+)\[\d+\]\s*=\s*\{(.*?)\};', text, re.S):
        arrays[name] = [int(n, 0) for n in re.findall(r'0x[\da-fA-F]+|(?<!\w)-?\d+', body)]
    clips = {}
    for name, frames, values, indices, static in re.findall(
        r'AnimationHeader\s+(\w+)\s*=\s*\{\s*\{\s*(\d+)\s*\},\s*(\w+),\s*(\w+),\s*(\d+)\s*\}', text):
        count, cutoff = int(frames), int(static)
        raw, joints = arrays[values], np.array(arrays[indices]).reshape(-1, 3)
        clip = np.array([[[raw[i if i < cutoff else i+f] for i in joint]
                         for joint in joints] for f in range(count)], dtype=float)
        clip[clip >= 32768] -= 65536
        label = name.removeprefix('gRomaniSkelGromani').removesuffix('animAnim')
        clips[label] = clip
    model = (HERE/'reference/romani_model.c.txt').read_text()
    limbs = []
    for x,y,z,child,sibling in re.findall(
        r'StandardLimb\s+\w+\s*=\s*\{\s*\{\s*(-?\d+),\s*(-?\d+),\s*(-?\d+)\s*\},\s*(\d+),\s*(\d+)', model):
        limbs.append(dict(pos=list(map(int,(x,y,z))),child=int(child),sibling=int(sibling)))
    assert len(limbs) == 22 and len(clips) == 10
    return limbs, clips

def hierarchy(limbs):
    parents = [-1] * len(limbs)
    def visit(i, parent):
        parents[i] = parent
        if limbs[i]['child'] != 255: visit(limbs[i]['child'], i)
        if limbs[i]['sibling'] != 255: visit(limbs[i]['sibling'], parent)
    visit(0, -1)
    return parents

def world_rotations(pose, parents):
    local = [rotation(v * np.pi/32768) for v in pose[1:]]
    result = [None] * len(parents)
    def get(i):
        if result[i] is None:
            result[i] = local[i] if parents[i] < 0 else get(parents[i]) @ local[i]
        return result[i]
    return np.array([get(i) for i in range(len(parents))])

def rotation(v):
    x,y,z = v
    sx,cx,sy,cy,sz,cz = math.sin(x),math.cos(x),math.sin(y),math.cos(y),math.sin(z),math.cos(z)
    return np.array([[cz*cy,cz*sy*sx-sz*cx,cz*sy*cx+sz*sx],
                     [sz*cy,sz*sy*sx+cz*cx,sz*sy*cx-cz*sx],[-sy,cy*sx,cy*cx]])

def angles(m):
    y = math.asin(float(np.clip(-m[2,0],-1,1)))
    if abs(math.cos(y)) > 1e-7:
        return np.array([math.atan2(m[2,1],m[2,2]),y,math.atan2(m[1,0],m[0,0])])
    return np.array([math.atan2(-m[1,2],m[1,1]),y,0])

def write_animation(template, frames, destination):
    # Dense channels avoid runtime adapters. Each joint has XYZ channels, one track per axis.
    nframes, njoints, _ = frames.shape
    values = np.round(frames).astype(np.int64)
    values = ((values + 32768) % 65536 - 32768).astype('<i2')
    tracks = values.transpose(1,2,0).reshape(-1)
    assert len(tracks) < 65536
    payload = bytearray(template[:64])
    payload += struct.pack('<IhI', 0, nframes, len(tracks)) + tracks.tobytes()
    payload += struct.pack('<I', njoints)
    for j in range(njoints): payload += struct.pack('<HHH', *(nframes*(j*3+a) for a in range(3)))
    payload += struct.pack('<H', 0)
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(payload)
    assert np.max(np.abs(load_animation(payload)-values)) == 0

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--archive', type=Path, required=True)
    parser.add_argument('--inspect', action='store_true')
    args = parser.parse_args()
    source, clips = source_data()
    source_parents = hierarchy(source)
    reference = world_rotations(clips['idle'][0], source_parents)
    archive = zipfile.ZipFile(args.archive)
    report = dict(source_commit=SOURCE_COMMIT, targets={})
    sockets = {}
    for age, obj, prefix in [('child','ma1','Child'), ('adult','ma2','Adult')]:
        skeleton = load_skeleton(archive, f'objects/object_{obj}/gMalon{prefix}Skel')
        original = archive.read(f'objects/object_{obj}/gMalon{prefix}IdleAnim')
        rest = load_animation(original)[0]
        parents = hierarchy(skeleton)
        if args.inspect:
            print(age, 'joints', len(rest))
            for i, limb in enumerate(skeleton): print(i+1, parents[i]+1, limb['name'].split('/')[-1], limb['pos'], rest[i+1].tolist())
            continue
        # Destination joint (including translation slot zero) -> source joint.
        mapping = ([0,1,2,3,4,5,6,7,8,13,9,13,17,18,19,20,21,22,14] if age == 'adult' else
                   [0,2,3,4,5,6,7,8,13,17,18,19,20,21,22,14,9,10,11])
        assert len(mapping) == len(rest), (age,len(mapping),len(rest))
        target_reference = world_rotations(rest, parents)
        hand = 11 if age == 'child' else 14
        sockets[age] = np.round(angles(target_reference[hand-1].T @ reference[18])*32768/np.pi).astype(int).tolist()
        root_scale = rest[0,1] / clips['idle'][0,0,1]
        outputs = {}
        for label, clip in clips.items():
            baked = []
            for pose in clip:
                source_world = world_rotations(pose, source_parents)
                desired = [source_world[src-1] @ reference[src-1].T @ target_reference[i]
                           for i,src in enumerate(mapping[1:])]
                target = np.zeros_like(rest)
                # Locomotion is actor-driven; preserve vertical animation, remove travelling root motion.
                target[0] = rest[0] + (pose[0]-clips['idle'][0,0])*root_scale
                target[0,0] = target[0,2] = 0
                for i, parent in enumerate(parents):
                    local = desired[i] if parent < 0 else desired[parent].T @ desired[i]
                    target[i+1] = angles(local) * 32768/np.pi
                baked.append(target)
            path = HERE/f'assets/animations/nei_malon/{age}/{label}'
            write_animation(original, np.array(baked), path)
            outputs[label] = dict(frames=len(baked), joints=len(rest))
        report['targets'][age] = dict(mapping=mapping,root_scale=root_scale,clips=outputs)
    if not args.inspect:
        socket_lines = ['// Baked hand orientation: target rest inverse * source rest.',
                        'static const s16 sBowSocket[2][3] = {']
        for age in ('adult','child'): socket_lines.append('    { '+', '.join(map(str,sockets[age]))+' },')
        socket_lines.append('};\n')
        (HERE/'bow_socket.inc.c').write_text('\n'.join(socket_lines))
        (HERE/'retarget-report.json.txt').write_text(json.dumps(report,indent=2)+'\n')
        print('Baked', sum(len(t['clips']) for t in report['targets'].values()), 'Malon clips')

if __name__ == '__main__': main()
